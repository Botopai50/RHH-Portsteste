from pathlib import Path
import zipfile

build_txt = r'''set -e

source /root/buildtools/harbourmasters/lib_build.sh

# Garante dependências básicas usadas durante o build.
apt-get update
apt-get install -y --no-install-recommends libssl1.1 python3

# Seu fork celshade otimizado.
# Se o workflow passar REF pela matriz/release, respeita o valor.
# Caso contrário, usa o branch do cel shading.
export REF="${REF:-wind-waker-style-cel-shading}"

project_clone https://github.com/Botopai50/ShipwrightTEST2.git

# --------------------------------------------------------------------
# Correção limpa de compatibilidade com libultraship novo.
#
# Problema 1:
# O código do ShipwrightTEST2 ainda tem vários arquivos chamando:
#   Ship::Context::GetRawInstance()
#
# Mas o libultraship atual só expõe:
#   Ship::Context::GetInstance()
#
# Correção:
#   GetRawInstance() -> GetInstance()
#
# Problema 2:
# Depois da troca, alguns arquivos passam a usar:
#   Ship::Context::GetInstance()->GetWindow()->GetGui()
#
# Porém alguns desses arquivos não incluem a definição completa de Ship::Window.
# Eles pegam só uma forward declaration vinda de Gui.h:
#   class Window;
#
# Aí o clang falha com:
#   member access into incomplete type 'Ship::Window'
#
# Correção:
#   adicionar #include <ship/window/Window.h>
#   em todo arquivo que usa Context::GetInstance()->GetWindow()
#
# Importante:
# - Não mexe no header do libultraship.
# - Não cria wrapper/gambiarra em Context.h.
# - Corrige o código fonte clonado antes de compilar.
# - Compatível com Python 3.8 do Ubuntu focal.
# --------------------------------------------------------------------
echo "Aplicando compatibilidade GetInstance/Window.h no código do SoH..."

python3 - <<'PY'
from pathlib import Path

root = Path("/root/build-port/project/soh")

if not root.exists():
    raise SystemExit(f"ERRO: diretório não encontrado: {root}")

extensions = {".cpp", ".hpp", ".h", ".cc", ".cxx"}
files = [p for p in root.rglob("*") if p.is_file() and p.suffix in extensions]

changed_getraw = []
added_window_include = []

def write_text_lf(path, text):
    # Compatível com Python 3.8.
    # Path.write_text(..., newline="\n") só existe em versões mais novas.
    with path.open("w", encoding="utf-8", newline="\n") as f:
        f.write(text)

for path in files:
    text = path.read_text(encoding="utf-8", errors="ignore")
    original = text

    # Atualiza chamadas antigas da API.
    text = text.replace("Ship::Context::GetRawInstance()", "Ship::Context::GetInstance()")
    text = text.replace("Context::GetRawInstance()", "Context::GetInstance()")

    if text != original:
        changed_getraw.append(path)

    # Se o arquivo acessa GetWindow(), precisa enxergar a definição completa de Ship::Window.
    uses_context_window = (
        "Ship::Context::GetInstance()->GetWindow()" in text
        or "Context::GetInstance()->GetWindow()" in text
    )

    has_window_include = (
        "#include <ship/window/Window.h>" in text
        or '#include "ship/window/Window.h"' in text
        or '#include <libultraship/window/Window.h>' in text
        or '#include "libultraship/window/Window.h"' in text
    )

    if uses_context_window and not has_window_include:
        lines = text.splitlines()

        insert_at = None

        # Preferimos colocar logo depois de ship/Context.h, se existir.
        for i, line in enumerate(lines):
            if "#include <ship/Context.h>" in line or '#include "ship/Context.h"' in line:
                insert_at = i + 1
                break

        # Caso não exista include direto de Context.h, coloca depois do último include inicial.
        if insert_at is None:
            last_include = None
            for i, line in enumerate(lines):
                stripped = line.strip()

                if stripped.startswith("#include"):
                    last_include = i
                    continue

                # Parou o bloco inicial de includes ao chegar em código real.
                if last_include is not None and stripped and not stripped.startswith("//"):
                    break

            if last_include is not None:
                insert_at = last_include + 1
            else:
                insert_at = 0

        lines.insert(insert_at, "#include <ship/window/Window.h>")
        text = "\n".join(lines) + "\n"
        added_window_include.append(path)

    if text != original:
        write_text_lf(path, text)

remaining_getraw = []
for path in files:
    text = path.read_text(encoding="utf-8", errors="ignore")
    if "GetRawInstance()" in text:
        remaining_getraw.append(path)

remaining_window_missing = []
for path in files:
    text = path.read_text(encoding="utf-8", errors="ignore")

    uses_context_window = (
        "Ship::Context::GetInstance()->GetWindow()" in text
        or "Context::GetInstance()->GetWindow()" in text
    )

    has_window_include = (
        "#include <ship/window/Window.h>" in text
        or '#include "ship/window/Window.h"' in text
        or '#include <libultraship/window/Window.h>' in text
        or '#include "libultraship/window/Window.h"' in text
    )

    if uses_context_window and not has_window_include:
        remaining_window_missing.append(path)

print(f"Arquivos com GetRawInstance() corrigidos: {len(changed_getraw)}")
for p in changed_getraw:
    print(f"  fixed-api: {p}")

print(f"Arquivos que receberam #include <ship/window/Window.h>: {len(added_window_include)}")
for p in added_window_include:
    print(f"  added-window-include: {p}")

print(f"Ocorrências restantes de GetRawInstance(): {len(remaining_getraw)}")
for p in remaining_getraw:
    print(f"  remaining-getraw: {p}")

print(f"Arquivos ainda usando GetWindow() sem Window.h: {len(remaining_window_missing)}")
for p in remaining_window_missing:
    print(f"  missing-window-include: {p}")

if remaining_getraw or remaining_window_missing:
    raise SystemExit("ERRO: patch de compatibilidade incompleto.")
PY

build_sdl2
build_sdl2_net
build_libzip
build_json
build_bzip2
build_tinyxml2
build_opus
build_opusfile
build_fmt
build_spdlog -DSPDLOG_FMT_EXTERNAL=ON

project_configure_and_build GenerateSohOtr \
    -DCMAKE_PREFIX_PATH=/usr/local \
    -DUSE_OPENGLES=1 \
    -DBUILD_REMOTE_CONTROL=1 \
    -DCMAKE_CXX_STANDARD=20 \
    -DCMAKE_CXX_STANDARD_REQUIRED=ON \
    -DCMAKE_C_FLAGS="-O3 -DNDEBUG -DPORTMASTER_CELSHADE_OPTIMIZED=1 -fno-math-errno -fno-trapping-math -ffunction-sections -fdata-sections -fno-semantic-interposition" \
    -DCMAKE_CXX_FLAGS="-O3 -DNDEBUG -DPORTMASTER_CELSHADE_OPTIMIZED=1 -DFMT_CONSTEVAL=constexpr -fno-math-errno -fno-trapping-math -ffunction-sections -fdata-sections -fno-semantic-interposition" \
    -DCMAKE_EXE_LINKER_FLAGS="-Wl,--gc-sections -Wl,--as-needed" \
    -DCMAKE_BUILD_TYPE=Release

ELF="$PROJECT_BUILD_DIR/soh/soh.elf"

echo "Dependências diretas do ELF:"
readelf -d "$ELF" | grep NEEDED || true

NEEDED_LIBS="$(readelf -d "$ELF" | sed -n 's/.*Shared library: \[\(.*\)\].*/\1/p')"
STAGE_LIBS=()

add_if_needed() {
    local lib="$1"

    if echo "$NEEDED_LIBS" | grep -Fxq "$lib"; then
        echo "stage_libs: adicionando dependência direta: $lib"
        STAGE_LIBS+=("$lib")
    else
        echo "stage_libs: pulando dependência indireta/não usada diretamente: $lib"
    fi
}

# Lista ampla de libs possíveis.
# Só entram no stage_libs se aparecerem diretamente no readelf NEEDED.
add_if_needed libz.so.1
add_if_needed libogg.so.0
add_if_needed libvorbis.so.0
add_if_needed libvorbisenc.so.2
add_if_needed libvorbisfile.so.3
add_if_needed libpng16.so.16
add_if_needed libbz2.so.1
add_if_needed libzip.so.5
add_if_needed libtinyxml2.so.10
add_if_needed libopus.so.0
add_if_needed libopusfile.so.0
add_if_needed libSDL2_net-2.0.so.0

if [ "${#STAGE_LIBS[@]}" -gt 0 ]; then
    stage_libs soh/soh.elf "${STAGE_LIBS[@]}"
else
    echo "AVISO: nenhuma lib direta extra foi encontrada para stage_libs."
fi

mkdir -p "$PROJECT_BUILD_DIR/libs"

copy_runtime_lib() {
    local lib="$1"
    local src=""

    if [ -f "$PROJECT_BUILD_DIR/libs/$lib" ]; then
        echo "runtime lib já existe: $lib"
        return 0
    fi

    for dir in \
        /usr/local/lib \
        /usr/lib/aarch64-linux-gnu \
        /lib/aarch64-linux-gnu \
        /usr/lib \
        /lib
    do
        if [ -e "$dir/$lib" ]; then
            src="$dir/$lib"
            break
        fi
    done

    if [ -z "$src" ]; then
        src="$(find /usr/local /usr /lib \( -type f -o -type l \) -name "$lib" 2>/dev/null | head -n 1 || true)"
    fi

    if [ -n "$src" ] && [ -e "$src" ]; then
        echo "Copiando runtime lib manual: $src -> $PROJECT_BUILD_DIR/libs/$lib"
        cp -L "$src" "$PROJECT_BUILD_DIR/libs/$lib"
    else
        echo "AVISO: runtime lib não encontrada para cópia manual: $lib"
    fi
}

# Copiadas manualmente porque podem ser dependências indiretas.
# O stage_libs rejeita libs que o ELF não NEED diretamente.
copy_runtime_lib libvorbis.so.0
copy_runtime_lib libvorbisenc.so.2
copy_runtime_lib libvorbisfile.so.3
copy_runtime_lib libogg.so.0
copy_runtime_lib libopus.so.0
copy_runtime_lib libopusfile.so.0

# OpenSSL 1.1 pode ser necessário indiretamente em alguns pacotes.
copy_runtime_lib libcrypto.so.1.1
copy_runtime_lib libssl.so.1.1

echo "Libs finais empacotadas:"
ls -lh "$PROJECT_BUILD_DIR/libs" || true

echo "Dependências diretas finais do ELF:"
readelf -d "$ELF" | grep NEEDED || true

echo "Build SoH celshade usando Botopai50/ShipwrightTEST2 concluído."
'''

out_dir = Path("/mnt/data/rhh_soh_buildtxt_python38_fix")
out_dir.mkdir(parents=True, exist_ok=True)
(out_dir / "build.txt").write_text(build_txt, encoding="utf-8")

zip_path = Path("/mnt/data/rhh_soh_buildtxt_python38_fix.zip")
with zipfile.ZipFile(zip_path, "w", zipfile.ZIP_DEFLATED) as z:
    z.write(out_dir / "build.txt", arcname="build.txt")

print(zip_path)
