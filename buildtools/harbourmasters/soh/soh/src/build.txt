set -e

# Keep the full surface-distance Texture LOD build script pinned to the previous commit, but patch a Python 3.8
# incompatibility before executing it. Python 3.8 pathlib.Path.write_text() does not support the newline= argument.
SCRIPT_URL="https://raw.githubusercontent.com/Botopai50/RHH-Portsteste/db8d2d84bd3e5cc39427f5d8ec5acc96c3502cf4/buildtools/harbourmasters/soh/soh/src/build.txt"
SCRIPT_PATH="/tmp/soh-build-with-surface-lod.sh"

if command -v curl >/dev/null 2>&1; then
    curl -fsSL "$SCRIPT_URL" -o "$SCRIPT_PATH"
elif command -v wget >/dev/null 2>&1; then
    wget -qO "$SCRIPT_PATH" "$SCRIPT_URL"
else
    echo "ERRO: curl/wget não encontrado para baixar o build script base"
    exit 1
fi

python3 - <<'PY'
from pathlib import Path

path = Path("/tmp/soh-build-with-surface-lod.sh")
text = path.read_text(encoding="utf-8")
old = 'path.write_text(text, encoding="utf-8", newline="\\n")'
new = '''with path.open("w", encoding="utf-8", newline="\\n") as f:
    f.write(text)'''

if old not in text:
    raise SystemExit("ERRO: chamada Path.write_text(..., newline=...) não encontrada no build script base")

text = text.replace(old, new, 1)
path.write_text(text, encoding="utf-8")
print("Compatibilidade Python 3.8 aplicada ao build script base.")
PY

bash "$SCRIPT_PATH"
