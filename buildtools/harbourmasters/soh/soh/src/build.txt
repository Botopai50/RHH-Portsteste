from pathlib import Path

root = Path("/root/build-port/project/libultraship")
header = root / "include/fast/backends/gfx_opengl.h"
cpp = root / "src/fast/backends/gfx_opengl.cpp"

for path in (header, cpp):
    if not path.exists():
        raise SystemExit(f"ERRO: arquivo não encontrado: {path}")

def write_text_lf(path, content):
    with path.open("w", encoding="utf-8", newline="\\n") as f:
        f.write(content)

h = header.read_text(encoding="utf-8")
old_h = '''struct TextureInfo {
    uint16_t width;
    uint16_t height;
    uint16_t filtering;
};'''
new_h = '''struct TextureInfo {
    uint16_t width;
    uint16_t height;
    uint16_t filtering;
    bool mipmapped = false;
};'''
if old_h not in h and "bool mipmapped = false;" not in h:
    raise SystemExit("ERRO: bloco TextureInfo não encontrado em gfx_opengl.h")
if old_h in h:
    h = h.replace(old_h, new_h, 1)
write_text_lf(header, h)

c = cpp.read_text(encoding="utf-8")

    glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, width, height, 0, GL_RGBA, GL_UNSIGNED_BYTE, rgba32_buf);
    textures[mCurrentTextureIds[mCurrentTile]].width = width;
    textures[mCurrentTextureIds[mCurrentTile]].height = height;
}'''
new_upload = '''static bool GfxShouldGenerateMipmaps(uint32_t width, uint32_t height) {
    // Mipmaps aumentam o uso de VRAM em ~33%, então não vale para texturas pequenas.
    // Para texturas HD grandes, reduzem bandwidth/cache miss quando vistas longe ou em ângulo.
    return width >= 64 || height >= 64;
}

void GfxRenderingAPIOGL::UploadTexture(const uint8_t* rgba32_buf, uint32_t width, uint32_t height) {
    glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA8, width, height, 0, GL_RGBA, GL_UNSIGNED_BYTE, rgba32_buf);

    TextureInfo& texture = textures[mCurrentTextureIds[mCurrentTile]];
    texture.width = width;
    texture.height = height;
    texture.mipmapped = GfxShouldGenerateMipmaps(width, height);

    if (texture.mipmapped) {
        glGenerateMipmap(GL_TEXTURE_2D);
    }
}'''
if old_upload not in c and "GfxShouldGenerateMipmaps" not in c:
    raise SystemExit("ERRO: função UploadTexture original não encontrada em gfx_opengl.cpp")
