root = Path("/root/build-port/project/soh")

def patch_virtual_destructors(text):
    text = text.replace(
        "class GameInteractionEffectBase {\n  public:\n    virtual GameInteractionEffectQueryResult CanBeApplied() = 0;",
        "class GameInteractionEffectBase {\n  public:\n    virtual ~GameInteractionEffectBase() = default;\n    virtual GameInteractionEffectQueryResult CanBeApplied() = 0;",

    text = text.replace(
        "class RemovableGameInteractionEffect : public GameInteractionEffectBase {\n  public:\n    virtual GameInteractionEffectQueryResult CanBeRemoved();",

    return text

        changed_getraw.append(path)

    before_destructor_patch = text
    if path.name == "GameInteractionEffect.h":
        text = patch_virtual_destructors(text)
    if text != before_destructor_patch:
        patched_virtual_destructors.append(path)

    uses_context_window = (

        or '#include "libultraship/window/Window.h"' in text

            last_include = None
            for i, line in enumerate(lines):
                stripped = line.strip()
                if stripped.startswith("#include"):
                    last_include = i
                    continue
                if last_include is not None and stripped and not stripped.startswith("//"):
                    break

static float ToonLodDot3(float ax, float ay, float az, float bx, float by, float bz) {
    return (ax * bx) + (ay * by) + (az * bz);
}

static void ClosestPointOnTriangleToPoint(float px, float py, float pz, float ax, float ay, float az, float bx,
                                          float by, float bz, float cx, float cy, float cz, float* outX,
                                          float* outY, float* outZ) {
    const float abx = bx - ax;
    const float aby = by - ay;
    const float abz = bz - az;
    const float acx = cx - ax;
    const float acy = cy - ay;
    const float acz = cz - az;
    const float apx = px - ax;
    const float apy = py - ay;
    const float apz = pz - az;

    const float d1 = ToonLodDot3(abx, aby, abz, apx, apy, apz);
    const float d2 = ToonLodDot3(acx, acy, acz, apx, apy, apz);
    if (d1 <= 0.0f && d2 <= 0.0f) {
        *outX = ax;
        *outY = ay;
        *outZ = az;
        return;
    }

    const float bpx = px - bx;
    const float bpy = py - by;
    const float bpz = pz - bz;
    const float d3 = ToonLodDot3(abx, aby, abz, bpx, bpy, bpz);
    const float d4 = ToonLodDot3(acx, acy, acz, bpx, bpy, bpz);
    if (d3 >= 0.0f && d4 <= d3) {
        *outX = bx;
        *outY = by;
        *outZ = bz;
        return;
    }

    const float vc = (d1 * d4) - (d3 * d2);
    if (vc <= 0.0f && d1 >= 0.0f && d3 <= 0.0f) {
        const float v = d1 / (d1 - d3);
        *outX = ax + (v * abx);
        *outY = ay + (v * aby);
        *outZ = az + (v * abz);
        return;
    }

    const float cpx = px - cx;
    const float cpy = py - cy;
    const float cpz = pz - cz;
    const float d5 = ToonLodDot3(abx, aby, abz, cpx, cpy, cpz);
    const float d6 = ToonLodDot3(acx, acy, acz, cpx, cpy, cpz);
    if (d6 >= 0.0f && d5 <= d6) {
        *outX = cx;
        *outY = cy;
        *outZ = cz;
        return;
    }

    const float vb = (d5 * d2) - (d1 * d6);
    if (vb <= 0.0f && d2 >= 0.0f && d6 <= 0.0f) {
        const float w = d2 / (d2 - d6);
        *outX = ax + (w * acx);
        *outY = ay + (w * acy);
        *outZ = az + (w * acz);
        return;
    }

    const float va = (d3 * d6) - (d5 * d4);
    if (va <= 0.0f && (d4 - d3) >= 0.0f && (d5 - d6) >= 0.0f) {
        const float w = (d4 - d3) / ((d4 - d3) + (d5 - d6));
        *outX = bx + (w * (cx - bx));
        *outY = by + (w * (cy - by));
        *outZ = bz + (w * (cz - bz));
        return;
    }

    const float denom = 1.0f / (va + vb + vc);
    const float v = vb * denom;
    const float w = vc * denom;
    *outX = ax + (abx * v) + (acx * w);
    *outY = ay + (aby * v) + (acy * w);
    *outZ = az + (abz * v) + (acz * w);
}

void Interpreter::UpdateDynamicTextureLodsForTriangle(const LoadedVertex* v1, const LoadedVertex* v2,
                                                      const LoadedVertex* v3, const ColorCombiner* comb) {
    if (!mDynamicLodEnabled || !mDynamicTextureLods || comb == nullptr || v1 == nullptr || v2 == nullptr ||
        v3 == nullptr) {
        return;
    }

    float surfaceX = 0.0f;
    float surfaceY = 0.0f;
    float surfaceZ = 0.0f;
    ClosestPointOnTriangleToPoint(mDynamicLodCamera[0], mDynamicLodCamera[1], mDynamicLodCamera[2], v1->wx, v1->wy,
                                  v1->wz, v2->wx, v2->wy, v2->wz, v3->wx, v3->wy, v3->wz, &surfaceX, &surfaceY,
                                  &surfaceZ);
    const uint8_t surfaceLod = SelectDynamicLodWorld(surfaceX, surfaceY, surfaceZ);

    for (int textureIndex = 0; textureIndex < 2; textureIndex++) {
        if (!comb->usedTextures[textureIndex]) {
            continue;
        }

        const uint32_t tile = mRdp->first_tile_index + textureIndex;
        if (tile >= 8) {
            continue;
        }

        const uint8_t tmemIndex = mRdp->texture_tile[tile].tmem_index;
        if (tmemIndex >= 2) {
            continue;
        }

        auto& loaded = mRdp->loaded_texture[tmemIndex];
        std::shared_ptr<Fast::Texture> texture = loaded.raw_tex_metadata.resource;
        if (texture == nullptr) {
            continue;
        }

        if ((loaded.tex_flags & TEX_FLAG_LOAD_AS_IMG) == 0) {
            continue;
        }

        const uint8_t* lodData = texture->GetLodData(surfaceLod);
        if (lodData == nullptr) {
            continue;
        }

        const uint16_t lodWidth = texture->GetLodWidth(surfaceLod);
        const uint16_t lodHeight = texture->GetLodHeight(surfaceLod);
        if (lodWidth == 0 || lodHeight == 0) {
            continue;
        }

        if (loaded.dynamic_lod == surfaceLod && loaded.addr == lodData && loaded.raw_tex_metadata.width == lodWidth &&
            loaded.raw_tex_metadata.height == lodHeight) {
            continue;

        loaded.addr = lodData;
        loaded.tex_flags = texture->Flags;
        loaded.raw_tex_metadata.width = lodWidth;
        loaded.raw_tex_metadata.height = lodHeight;
        loaded.raw_tex_metadata.h_byte_scale = texture->HByteScale;
        loaded.raw_tex_metadata.v_pixel_scale = texture->VPixelScale;
        loaded.raw_tex_metadata.type = texture->Type;
        loaded.raw_tex_metadata.resource = texture;
        loaded.dynamic_lod = surfaceLod;
        mRdp->textures_changed[tmemIndex] = true;

}

copy_runtime_lib() {
    local lib="$1"
    local src=""

    if [ -f "$PROJECT_BUILD_DIR/libs/$lib" ]; then
        echo "runtime lib já existe: $lib"
        return 0
    fi

    for dir in \
        /usr/local/lib \
        /usr/lib/aarch64-linux-gnu \
        /lib/aarch64-linux-gnu \
        /usr/lib \
        /lib
    do
        if [ -e "$dir/$lib" ]; then
            src="$dir/$lib"
            break
        fi
    done

    if [ -z "$src" ]; then
        src="$(find /usr/local /usr /lib \( -type f -o -type l \) -name "$lib" 2>/dev/null | head -n 1 || true)"
    fi

    if [ -n "$src" ] && [ -e "$src" ]; then
        echo "Copiando runtime lib manual: $src -> $PROJECT_BUILD_DIR/libs/$lib"
        cp -L "$src" "$PROJECT_BUILD_DIR/libs/$lib"
    else
        echo "AVISO: runtime lib não encontrada para cópia manual: $lib"
    fi
}
